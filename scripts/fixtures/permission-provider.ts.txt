const core=createFauxCore({provider:'permission-fixture',models:[{id:'reviewer-fixture'}]});

     const isReview=text.includes('"evidence"')&&text.includes('"request"');

     return fauxAssistantMessage([fauxText('Fixture finished.')]);
   }]);return core.streamSimple(model,context,options);
 }});
